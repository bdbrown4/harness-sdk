cmake_minimum_required(VERSION 3.16)
if(POLICY CMP0141)
  cmake_policy(SET CMP0141 NEW)
endif()
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
set(CMAKE_TRY_COMPILE_CONFIGURATION Release)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(AWS_STATIC_MSVC_RUNTIME_LIBRARY ON CACHE BOOL "" FORCE)
project(StrandsNovaCrtProof LANGUAGES C)

set(IN_SOURCE_BUILD ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/source/aws-c-common/cmake")
include(AwsFindPackage)

add_subdirectory(source/aws-c-common deps/common)
add_subdirectory(source/aws-c-io deps/io)
add_subdirectory(source/aws-c-sdkutils deps/sdkutils)
add_subdirectory(source/aws-c-cal deps/cal)
add_subdirectory(source/aws-c-compression deps/compression)
add_subdirectory(source/aws-c-http deps/http)
add_subdirectory(source/aws-c-auth deps/auth)
add_subdirectory(source/aws-checksums deps/checksums)

add_library(strands-nova-crt SHARED native/strands_nova_crt.c native/http_duplex.c native/strands_nova_crt.def)
target_link_libraries(strands-nova-crt PRIVATE aws-c-auth)
target_compile_options(strands-nova-crt PRIVATE /W4 /WX /Z7 /Gd)
set_target_properties(strands-nova-crt PROPERTIES C_STANDARD 99)
